cmake_minimum_required(VERSION 3.16)
project(outfit-swap-dll CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(outfit-swap-dll SHARED outfit-swap-dll.cpp)
target_include_directories(outfit-swap-dll PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(outfit-swap-dll PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(outfit-swap-dll PRIVATE /W3 /MT)
    set_property(TARGET outfit-swap-dll PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
set_target_properties(outfit-swap-dll PROPERTIES
    PREFIX "" OUTPUT_NAME "outfit-swap-dll")
