cmake_minimum_required(VERSION 3.16)
project(probe-comp-crash CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(probe-comp-crash SHARED probe.cpp)
target_include_directories(probe-comp-crash PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(probe-comp-crash PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(probe-comp-crash PRIVATE /W3 /MT)
    set_property(TARGET probe-comp-crash PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_libraries(probe-comp-crash PRIVATE psapi)
endif()
set_target_properties(probe-comp-crash PROPERTIES PREFIX "" OUTPUT_NAME "probe-comp-crash")
