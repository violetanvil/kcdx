cmake_minimum_required(VERSION 3.16)
project(comp-03-B CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(comp-03-B SHARED comp-03-B.cpp)
target_include_directories(comp-03-B PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(comp-03-B PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(comp-03-B PRIVATE /W3 /MT)
    set_property(TARGET comp-03-B PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_libraries(comp-03-B PRIVATE psapi)
endif()
set_target_properties(comp-03-B PROPERTIES PREFIX "" OUTPUT_NAME "comp-03-B")
