cmake_minimum_required(VERSION 3.16)
project(cap-01 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(cap-01 SHARED cap-01.cpp)
target_include_directories(cap-01 PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(cap-01 PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(cap-01 PRIVATE /W3 /MT)
    set_property(TARGET cap-01 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_libraries(cap-01 PRIVATE psapi)
endif()
set_target_properties(cap-01 PROPERTIES PREFIX "" OUTPUT_NAME "cap-01")
