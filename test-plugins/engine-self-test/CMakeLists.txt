cmake_minimum_required(VERSION 3.16)
project(engine-self-test CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(engine-self-test SHARED engine-self-test.cpp)
target_include_directories(engine-self-test PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(engine-self-test PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(engine-self-test PRIVATE /W3 /MT)
    set_property(TARGET engine-self-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
set_target_properties(engine-self-test PROPERTIES
    PREFIX "" OUTPUT_NAME "engine-self-test")
