cmake_minimum_required(VERSION 3.16)
project(comp-02 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(comp-02 SHARED comp-02.cpp)
target_include_directories(comp-02 PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(comp-02 PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(comp-02 PRIVATE /W3 /MT)
    set_property(TARGET comp-02 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_libraries(comp-02 PRIVATE psapi)
endif()
set_target_properties(comp-02 PROPERTIES PREFIX "" OUTPUT_NAME "comp-02")
