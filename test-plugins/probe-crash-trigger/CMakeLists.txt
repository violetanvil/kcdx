cmake_minimum_required(VERSION 3.16)
project(probe-crash-trigger CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED KCDX_INCLUDE_DIR)
    set(KCDX_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include")
endif()
add_library(probe-crash-trigger SHARED probe.cpp)
target_include_directories(probe-crash-trigger PRIVATE "${KCDX_INCLUDE_DIR}")
if(MSVC)
    target_compile_definitions(probe-crash-trigger PRIVATE
        _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(probe-crash-trigger PRIVATE /W3 /MT)
    set_property(TARGET probe-crash-trigger PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
set_target_properties(probe-crash-trigger PROPERTIES PREFIX "" OUTPUT_NAME "probe-crash-trigger")
